<template>
  <div class="modal-overlay">
    <div class="modal-content">

      <!-- ヘッダー：キャラクター切替 -->
      <div class="character-header">
        <button @click="prevCharacter" class="arrow-btn">〈</button>

        <div class="character-info">
          <img :src="getRollIcon(currentRace)" alt="種族" class="icon" />
          <img :src="getRollIcon(currentClass)" alt="クラス" class="icon" />
          <span class="name">{{ currentCharacter.name }}</span>
          <span class="level-tag">Lv {{ currentCharacter.stats.allLv }}</span>
        </div>

        <button @click="nextCharacter" class="arrow-btn">〉</button>
      </div>

      <!-- タブバー -->
      <div class="tab-bar">
        <div
          v-for="tab in tabs"
          :key="tab"
          :class="['tab-btn', { active: currentTab === tab }]"
          @click="currentTab = tab"
        >
          <div class="tab-icon">🜛</div>
          <div class="tab-label">{{ tab }}</div>
        </div>
      </div>

      <!-- タブ内容 -->
      <div class="tab-content">
        <!-- 基本 -->
        <div v-if="currentTab === '基本'" class="tab-panel">
          <p>名前: {{ currentCharacter.name }}</p>
          <p>種族: {{ currentRace }}</p>
          <p>クラス: {{ currentClass }}</p>
          <p>Lv: {{ currentCharacter.stats.allLv }}</p>
          <p>所持金: {{ character.money }}</p>
          <p>現在地: {{ character.location }}</p>
        </div>

        <!-- ステータス -->
        <div v-if="currentTab === 'ステータス'" class="tab-panel">
          <div v-for="(role, idx) in currentCharacter.Role" :key="idx" v-if="role.Lv > 0">
            {{ role.roleName }} Lv{{ role.Lv }}
          </div>
          <div class="stats-grid">
            <div v-for="(val, key) in currentCharacter.stats.baseStats" :key="key">
              {{ key }}: {{ val }}
            </div>
          </div>
        </div>

        <!-- 技 -->
        <div v-if="currentTab === '技'" class="tab-panel">
          <div v-for="s in currentCharacter.skills" :key="s.名前">
            {{ s.名前 }} ({{ s.行動 }})
          </div>
          <p>属性: {{ currentCharacter.attribute }}</p>
        </div>

        <!-- 装備 -->
        <div v-if="currentTab === '装備'" class="tab-panel">
          <div v-for="(val, key) in currentCharacter.equipmentSlot" :key="key">
            {{ key }}: {{ val || 'なし' }}
          </div>
        </div>

        <!-- 手持ち -->
        <div v-if="currentTab === '手持ち'" class="tab-panel">
          <p>所持上限: {{ character.maxInventory }}</p>
          <ul>
            <li v-for="(item, idx) in character.inventory" :key="idx">
              {{ item.name }} × {{ item.quantity }}
            </li>
          </ul>
        </div>

        <!-- ギルド -->
        <div v-if="currentTab === 'ギルド'" class="tab-panel">
          <p>所属: {{ character.guild.name }}</p>
          <p>ランク: {{ character.guild.rank }}</p>
          <p>貢献度: {{ character.guild.contributionPoints }}</p>
        </div>

        <!-- 進行 -->
        <div v-if="currentTab === '進行'" class="tab-panel">
          <p>転移可能な街:</p>
          <ul>
            <li v-for="town in character.memoryStreet" :key="town">{{ town }}</li>
          </ul>
          <p>クエスト進行:</p>
          <ul>
            <li v-for="q in character.questProgress" :key="q.id">
              {{ q.id }}: {{ q.state }}
            </li>
          </ul>
        </div>
      </div>

      <!-- フッター -->
      <div class="footer">
        <button class="btn confirm" @click="ok">冒険を続ける</button>
        <button class="btn cancel" @click="close">選択に戻る</button>
      </div>
    </div>
  </div>
</template>

<script>
import { ref, computed } from "vue";
import { getRollIcon } from "@/constants/statData";

export default {
  name: "CharacterStatusModal",
  props: {
    character: { type: Object, required: true }, // playerGlobalData
  },
  emits: ["close", "ok"],
  setup(props, { emit }) {
    const index = ref(0);
    const currentTab = ref("基本");
    const tabs = ["基本", "ステータス", "技", "装備", "手持ち", "ギルド", "進行"];

    const party = computed(() => props.character?.party ?? []);
    const currentCharacter = computed(() => {
      const c = party.value[index.value];
      if (!c) return { name: "不明", stats: { allLv: 0, baseStats: {} }, Role: [] };
      return { stats: { allLv: 0, baseStats: {} }, ...c };
    });

    const currentRace = computed(() => currentCharacter.value?.Role?.[1]?.roleName || "");
    const currentClass = computed(() => currentCharacter.value?.Role?.[0]?.roleName || "");

    const prevCharacter = () => { if (index.value > 0) index.value--; };
    const nextCharacter = () => { if (index.value < party.value.length - 1) index.value++; };

    return {
      index, tabs, currentTab,
      party, currentCharacter, currentRace, currentClass,
      getRollIcon,
      prevCharacter, nextCharacter,
      close: () => emit("close"),
      ok: () => emit("ok", currentCharacter.value),
    };
  },
};
</script>
